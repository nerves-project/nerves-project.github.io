{{- /* The tag chips above an article list: "all" plus the most used article tags, with counts. Usage: partial "articles/chips" (dict "active" "all") or (dict "active" .Data.Term) on a tag page. "nerves" is left out (nearly every article has it). The active tag is added when it is not among the top ones, so a tag page always shows its own chip. */ -}} {{- $active := .active | default "all" -}} {{- $limit := 8 -}} {{- $section := site.GetPage "/articles" -}} {{- $all := where site.RegularPages "Section" "articles" -}} {{- $chips := slice -}} {{- range site.Taxonomies.tags.ByCount -}} {{- $n := len (where .Pages "Section" "articles") -}} {{- if and (gt $n 0) (ne .Name "nerves") -}} {{- $chips = $chips | append (dict "name" .Name "title" .Page.LinkTitle "url" .Page.RelPermalink "n" $n) -}} {{- end -}} {{- end -}} {{- $chips = first $limit (sort $chips "n" "desc") -}} {{- if and (ne $active "all") (not (where $chips "name" $active)) -}} {{- with site.GetPage (printf "/tags/%s" $active) -}} {{- $chips = $chips | append (dict "name" $active "title" .LinkTitle "url" .RelPermalink "n" (len .Pages)) -}} {{- end -}} {{- end -}}
  • all · {{ len $all }}
    • {{- range $chips }}
  • {{ .title }} · {{ .n }}
    • {{- end }}
