{{- /* A short plain-text excerpt for cards and the RSS feed. Usage: partial "blog/excerpt.html" (dict "page" . "length" 200) 1. the description front matter, when there is one (used as written); 2. for a newsletter issue: the first real paragraph (after the first heading, for an issue that is still the raw email HTML), cut to `length`. Never .Summary: for a raw email that is the whole email, and for a cleaned issue it runs the section heading into the first sentence; 3. otherwise the summary of the post, cut to `length`. */ -}} {{- $p := .page -}} {{- $len := .length | default 200 -}} {{- $out := "" -}} {{- $raw := findRE `(?i)]*>(.*?)

` $body 5 -}} {{- $t := index . 1 | plainify | htmlUnescape | strings.TrimSpace -}} {{- if and (not $out) (gt (len $t) 40) }}{{ $out = $t }}{{ end -}} {{- end -}} {{- else -}} {{- $out = $p.Summary | plainify | htmlUnescape -}} {{- end -}} {{- /* truncate escapes HTML characters, so unescape again: callers escape for their output */ -}} {{- $out = replaceRE `\s+` " " $out | strings.TrimSpace | truncate $len | htmlUnescape -}} {{- end -}} {{- return $out -}}
